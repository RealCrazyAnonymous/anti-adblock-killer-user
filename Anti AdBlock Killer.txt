! ============================================================
! SECTION 3: EXCEPTION RULES – ALLOW BAIT ELEMENTS / SCRIPTS
! ============================================================
! Anti‑adblock scripts often check for the presence of a “bait” element
! (e.g., a div with class “adsbox” or a script named “show_ads.js”).
! If the bait is missing, the site concludes an ad blocker is active.
! Exception rules allow these specific baits to load so the check passes.

! --- Allow common bait scripts ---
@@||example.com/js/show_ads.js$script
@@||example.com/ads/bait.js$script
@@||example.com/ads.js$script

! --- Allow bait HTML elements (so they are not hidden by cosmetic filters) ---
example.com#@#.adsbox
example.com#@#.ad-banner
example.com#@#.ad-slot
example.com#@#div[id="ad-bait"]

! --- Generic exceptions for ubiquitous bait class names ---
#@#.adsbox
#@#.ad-banner
#@#.ad-slot

! ============================================================
! SECTION 4: SCRIPTLET INJECTION – NEUTRALIZE DETECTION LOGIC
! ============================================================
! Scriptlets are small JavaScript snippets injected by uBlock Origin
! (and compatible blockers) to patch page‑side anti‑adblock checks.
! The syntax is:  domain##+js(scriptlet-name, argument1, argument2)
! These rules require uBlock Origin or a blocker that supports scriptlets.

! --- Override common global variables used for detection ---
example.com##+js(set, adsbygoogle, [])
example.com##+js(set, adBlockDetected, false)
example.com##+js(set, canRunAds, true)
example.com##+js(set, showAds, true)
example.com##+js(set, adblock, false)
example.com##+js(set, isAdBlockActive, false)

! --- Neutralize session/local storage flags ---
example.com##+js(set-session-storage-item, adblock_shown, false)
example.com##+js(set-local-storage-item, adblock_detected, false)

! --- Generic scriptlet rules that work on many sites ---
##+js(set, Object.prototype.adBlocksFound, false)
##+js(set, Object.prototype.hasAllowedInstreamAd, true)

! --- YouTube‑style anti‑adblock (example) ---
youtube.com##+js(set, yt.config_.openPopupConfig.supportedPopups.adBlockMessageViewModel, false)
youtube.com##+js(set, Object.prototype.adBlocksFound, false)

! --- Remove “nag” cookies / storage that trigger repeated warnings ---
example.com##+js(remove-cookie, adblock_warning)
example.com##+js(remove-cookie, anti_adblock)

! ============================================================
! SECTION 5: $REPLACE RULES – PATCH ANTI‑ADBLOCK RESPONSES
! ============================================================
! The $replace modifier (supported by uBlock Origin 1.53+ and AdGuard)
! can rewrite the body of a response. Use it to strip detection code
! from HTML or JavaScript responses when blocking the whole file would
! break the site.

! --- Example: remove anti‑adblock snippet from an HTML response ---
||example.com^$replace=/<script[^>]*anti-adblock[^>]*>.*?<\/script>//g

! --- Example: neutralize a specific detection variable in a JS file ---
||example.com/detect.js$replace=/adBlockDetected\s*=\s*true/adBlockDetected = false/g

! --- Example: remove a “wall” div from the HTML before it renders ---
||example.com^$replace=/<div[^>]*class="[^"]*adblock-wall[^"]*"[^>]*>.*?<\/div>//g
